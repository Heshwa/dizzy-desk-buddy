cmake_minimum_required(VERSION 3.18)
project(dizzy_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE FILEPATH "")
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "")
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "")
if(NOT DEFINED LVGL_SOURCE)
  set(LVGL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.work/lvgl")
endif()
add_subdirectory("${LVGL_SOURCE}" lvgl)
find_package(SDL2 REQUIRED CONFIG)
add_executable(dizzy-preview main.cpp ../shared/face.cpp)
target_include_directories(dizzy-preview PRIVATE ../shared)
target_link_libraries(dizzy-preview PRIVATE lvgl SDL2::SDL2)
